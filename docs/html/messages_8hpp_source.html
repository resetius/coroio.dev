32template<typename TLambda>
33struct TLambdaToContextDeleter {
34 static_assert(
35 std::is_empty_v<TLambda> || sizeof(TLambda) <= sizeof(void*),
36 "Lambda must be stateless or capture only one pointer-sized value");
37
38 // Stateless lambda to context deleter conversion
39 template<typename F>
40 static std::enable_if_t<std::is_empty_v<std::decay_t<F>>, TContextDeleter>
41 Convert(F&& lambda) {
42 auto func_ptr = +[](void* /*unused*/, void* ptr) {
43 std::decay_t<F>{}(ptr);
44 };
45 return TContextDeleter{func_ptr, nullptr};
46 }
47
48 template<typename F>
49 static std::enable_if_t<!std::is_empty_v<std::decay_t<F>>, TContextDeleter>
50 Convert(F&& lambda) {
51 void* captured_ptr = ExtractCapturedPointer(lambda);
52
53 auto func_ptr = +[](void* ctx, void* ptr) {
54 using LambdaType = std::decay_t<F>;
55 alignas(LambdaType) char buffer[sizeof(LambdaType)];
56 *reinterpret_cast<void**>(buffer) = ctx;
57 auto* lambda_ptr = reinterpret_cast<LambdaType*>(buffer);
58 (*lambda_ptr)(ptr);
59 };
60
61 return TContextDeleter{func_ptr, captured_ptr};
62 }
63
64private:
65 template<typename F>
66 static void* ExtractCapturedPointer(const F& lambda) {
67 return *reinterpret_cast<void* const*>(&lambda);
68 }
69};

160typename std::enable_if_t<is_pod_v<T>, TBlob>
161SerializePodNear(TAllocator& alloc, Args&&... args)
162{
163 constexpr uint32_t size = sizeof_data<T>();
164
165 TBlob::TRawPtr rawPtr = nullptr;
166

169 new (data) T(std::forward<Args>(args)...);

269template<typename T>
270TBlob SerializeFar(TBlob blob)
271{
272 if constexpr (is_pod_v<T>) {
273 blob.Type = TBlob::PointerType::Far;
274 return blob; // For POD, far == near, just share the pointer
275 } else {
276 T* obj = reinterpret_cast<T*>(blob.Data.get());
277 const uint32_t size = FarPayloadSize(*obj);
278 auto rawPtr = TBlob::TRawPtr(::operator new(size), TBlobDeleter{[](void* ptr) {
279 ::operator delete(ptr);
280 }});
281 SerializeTo(*obj, std::span<char>(static_cast<char*>(rawPtr.get()), size));
282 return TBlob{std::move(rawPtr), size, TBlob::PointerType::Far};
283 }

303template<typename T, typename TStream, typename... Args>
304void SerializeFarInplace(TStream& stream, TActorId sender, TActorId recipient, Args&&... args)
305{
306 if constexpr (is_pod_v<T>) {
307 constexpr auto size = sizeof_data<T>() + sizeof(THeader);
308 auto buf = stream.Acquire(size);
309 char* p = static_cast<char*>(buf.data());
310 // The buffer position follows the previous payload, so it may be unaligned for THeader.
311 const THeader header{sender, recipient, T::MessageId, sizeof_data<T>()};
312 std::memcpy(p, &header, sizeof(header));
313 p += sizeof(THeader);
314 // TODO: this position may be unaligned for T too (UB by the standard, works on
315 // x86-64/arm64); construct T aside and memcpy it, as the header above.
316 new (p) T(std::forward<Args>(args)...);
317 stream.Commit(size);
318 } else {
319 const T obj(std::forward<Args>(args)...);
320 const uint32_t payload = FarPayloadSize(obj);
321 const size_t size = sizeof(THeader) + payload;
322 auto buf = stream.Acquire(size);
323 const THeader header{sender, recipient, T::MessageId, payload};
324 std::memcpy(buf.data(), &header, sizeof(header));
325 SerializeTo(obj, std::span<char>(buf.data() + sizeof(THeader), payload));
326 stream.Commit(size);
327 }
328}
