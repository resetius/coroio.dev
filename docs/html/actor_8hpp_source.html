539 messageId,
540 blob,
541 ctx
542 ) || ...);
543
544 if (!handled) {
545 static_cast<TBaseBehavior*>(this)->HandleUnknownMessage(messageId, std::move(blob), std::move(ctx));
546 }

549private:
550 template<typename TMessage>
551 bool TryHandleMessage(TMessageId messageId, TBlob& blob, TActorContext::TPtr& ctx) {
552 if (TMessage::MessageId == messageId) {
553 if (blob.Type == TBlob::PointerType::Near) {
554 auto&& mes = DeserializeNear<TMessage>(blob);
555 HandleMessage<TMessage>(
556 std::move(mes),
557 std::move(blob),
558 std::move(ctx)
559 );
560 } else if constexpr (is_local_message_v<TMessage>) {
561 return false;
562 } else {
563 auto&& mes = DeserializeFar<TMessage>(blob);
564 HandleMessage<TMessage>(
565 std::move(mes),
566 std::move(blob),
567 std::move(ctx)
568 );
569 }
570 return true;
571 }
572 return false;
573 }
574
575 template<typename TMessage>
576 void HandleMessage(TMessage&& message, TBlob blob, TActorContext::TPtr ctx)
577 {
578 using ReturnType = decltype(static_cast<TBaseBehavior*>(this)->Receive(
579 std::declval<TMessage>(),
580 std::declval<TBlob>(),
581 std::declval<TActorContext::TPtr>()
582 ));
583
584 if constexpr (std::is_same_v<ReturnType, void>) {
585 static_cast<TBaseBehavior*>(this)->Receive(
586 std::move(message),
587 std::move(blob),
588 std::move(ctx)
589 );
590 } else {
591 auto async = ctx->StartAsync();
592 auto future = static_cast<TBaseBehavior*>(this)->Receive(
593 std::move(message),
594 std::move(blob),
595 std::move(ctx)
596 );
597 if (!future.done()) {
598 async.Commit(std::move(future));
599 }
