101 void Send(TEnvelope&& envelope) override {
102 // Empty blobs skip the factory and its compile-time check of the id
103 if (envelope.MessageId > MaxMessageId) [[unlikely]] {
104 throw std::invalid_argument("message id " + std::to_string(envelope.MessageId)
105 + " overlaps the padding bits of THeader::MessageId");
106 }
107 auto blob = std::move(envelope.Blob);
108 uint32_t padding = 0;
109 // TODO: a blob already in Far form (a received non-POD message Forward-ed to another
110 // node) is serialized again as if it held a T object; pass such blobs through as bytes.
111 if (blob.Size > 0) {
112 blob = Factory.SerializeFar(envelope.MessageId, std::move(blob));
113 const size_t alignment = Factory.PayloadAlignment(envelope.MessageId);
114 if (alignment > 1) {
115 padding = PayloadPadding(Offset(), alignment);
116 }
117 }
118 THeader data{
119 .Sender = envelope.Sender,
120 .Recipient = envelope.Recipient,
121 .MessageId = envelope.MessageId | (padding << HeaderPaddingShift),
122 .Size = blob.Size
123 };
124 const size_t size = sizeof(data) + padding + blob.Size;
125 auto buf = Acquire(size);
126 std::memcpy(buf.data(), &data, sizeof(data));
127 std::memset(buf.data() + sizeof(data), 0, padding);
128 if (blob.Size > 0) {
129 std::memcpy(buf.data() + sizeof(data) + padding, blob.Data.get(), blob.Size);
130 }
131 Commit(size);
132 }

172 TFuture<void> DoDrain() {
173 try {
174 while (CommittedBytes > 0) {
175 SendBuffer.Swap(OutputBuffer);
176 auto readSize = CommittedBytes;
177 CommittedBytes = 0;
178 SentBytes += readSize;
179 co_await TByteWriter(Socket).Write(SendBuffer.Data(), readSize);
180 }
181 co_return;
182 } catch (const std::exception& ex) {
183 std::cerr << "Error during draining: " << ex.what() << "\n";
184 Connect();
185 co_return;
186 }

189 void Connect() {
190 if (Connector.raw() && !Connector.done()) {
191 return;
192 }
193
194 Connector = DoConnect();
195 }

197 TFuture<void> DoConnect() {
198 Connected = false;
199 // Without aligned frames the new stream starts at 0 with no filler, as before PayloadAlignment
200 if (!WroteAligned) {
201 SentBytes = 0;
202 }
203 std::cout << "Connecting to " << HostPort.ToString() << "\n";
204 while (!Connected) {
205 try {
206 //auto deadline = NNet::TClock::now() + std::chrono::milliseconds(5000);
207 TAddress addr = co_await HostPort.Resolve(Resolver);
208 Socket = SocketFactory(addr);
209 // Small messages (acks) must not wait for Nagle and delayed ACK.
210 Socket.SetNoDelay();
211 co_await Socket.Connect(addr /*, deadline*/);
212 // The queued frames are padded for the old stream offset: the filler gives them
213 // the same offsets mod MaxPayloadAlignment in the new stream, and they stay as is.
214 char filler[MaxFillerSize];
215 if (const size_t size = WriteFiller(filler, SentBytes)) {
216 co_await TByteWriter(Socket).Write(filler, size);
217 }
218 Connected = true;
219 std::cout << "Connected to " << HostPort.ToString() << "\n";
220 } catch (const std::exception& ex) {
221 std::cerr << "Error connecting to " << HostPort.ToString() << ": " << ex.what() << "\n";
222 Connected = false;
223 }
224 if (!Connected) {
225 co_await Poller.Sleep(std::chrono::milliseconds(1000));
226 }
227 }
228 // Nothing calls Drain() again for bytes queued while connecting (or left after a send error).
229 if (CommittedBytes > 0) {
230 StartDrain();
231 }
232 co_return;
233 }
