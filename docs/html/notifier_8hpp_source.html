126 struct TWaitGuard {
127 TNotifier& Owner;
128 bool Completed = false;
129 ~TWaitGuard() {
130 Owner.Waiting_ = false;
131 if constexpr (!std::is_same_v<TFileHandle, NNet::TFileHandle>) {
132 // A cancelled driven read may have consumed the signal already;
133 // signal again so the notification is retained. Worst case is a spurious wakeup.
134 if (!Completed && Owner.Pending_.load(std::memory_order_acquire)) {
135 try {
136 NDetail::WriteNotifierSignal(Owner.WriteFd_);
137 } catch (...) {
138 }
139 }
140 }
141 }
142 } guard{*this};
143
144#ifdef __linux__
145 uint64_t signal;
146#else
147 uint8_t signal;
148#endif
149 for (;;) {
150 const auto size = co_await Handle_.ReadSome(&signal, sizeof(signal));
151 if (size == static_cast<decltype(size)>(sizeof(signal))) {
152 break;
153 }
154 if (size >= 0) {
155 throw std::system_error(EIO, std::generic_category(), "notifier read");
156 }
157 }
158 // Re-arm before the caller drains its work, so concurrent work can signal again.
159 Pending_.exchange(false, std::memory_order_acq_rel);
160 guard.Completed = true;
161 }
