38 constexpr auto messageId = T::MessageId;
39 static_assert(messageId <= MaxMessageId, "the top bits of THeader::MessageId hold the payload padding");
40 static_assert(messageId != FillerMessageId, "the id of the filler frame is reserved");

45 return ::NNet::NActors::SerializeFar<T>(std::move(blob));
