131 // The framing is the one of TNode::Send; a deferred payload is left to the I/O side
132 void Send(TEnvelope&& envelope) override {
133 // Empty blobs skip the factory and its compile-time check of the id
134 if (envelope.MessageId > MaxMessageId) [[unlikely]] {
135 throw std::invalid_argument("message id " + std::to_string(envelope.MessageId)
136 + " overlaps the padding bits of THeader::MessageId");
137 }
138 std::lock_guard guard(Lock_);
139 auto blob = std::move(envelope.Blob);
140 uint32_t padding = 0;
141 uint32_t payload = 0;
142 const TMessagesFactory::TPayloadWriter* writer = nullptr;
143 if (blob.Size > 0) {
144 if (blob.Type == TBlob::PointerType::Near) {
145 writer = Factory_.PayloadWriter(envelope.MessageId);
146 }
147 if (writer) {
148 // Sized here: an invalid object throws to the sender, as with TNode
149 payload = writer->Size(blob);
150 } else {
151 blob = Factory_.SerializeFar(envelope.MessageId, std::move(blob));
152 payload = blob.Size;
153 }
154 const size_t alignment = Factory_.PayloadAlignment(envelope.MessageId);
155 if (alignment > 1) {
156 padding = PayloadPadding(Offset(), alignment);
157 }
158 }
159 THeader data{
160 .Sender = envelope.Sender,
161 .Recipient = envelope.Recipient,
162 .MessageId = envelope.MessageId | (padding << HeaderPaddingShift),
163 .Size = payload
164 };
165 const size_t size = sizeof(data) + padding + payload;
166 auto buf = Acquire(size);
167 std::memcpy(buf.data(), &data, sizeof(data));
168 std::memset(buf.data() + sizeof(data), 0, padding);
169 if (writer) {
170 Current_->Deferred.push_back(TDeferredPayload{
171 .Offset = Current_->Size + sizeof(data) + padding,
172 .Size = payload,
173 .Write = writer->Write,
174 .Blob = std::move(blob),
175 });
176 } else if (payload > 0) {
177 std::memcpy(buf.data() + sizeof(data) + padding, blob.Data.get(), payload);
178 }
179 Commit(size);
180 }
181
182 std::span<char> Acquire(size_t size) override {
183 auto& segment = *Current_;
184 segment.Data.Reserve(segment.Size, segment.Size + size);
185 return std::span<char>(segment.Data.Data() + segment.Size, size);
186 }
187
188 void Commit(size_t size) override {
189 Current_->Size += size;
190 }
191
192 uint64_t Offset() const override {
193 if (!WroteAligned_) [[unlikely]] {
194 // One atomic step against the CAS in Connect(): either that moved the base first
195 // and this reads the new one, or it sees the bit and keeps the base. The word
196 // carries all the two sides share here, so nothing is ordered by it: relaxed.
197 Base_ = AlignState_.fetch_or(AlignedBit, std::memory_order_relaxed) >> 1;
198 WroteAligned_ = true;
199 }
200 return SealedBytes_ + Current_->Size - Base_;
201 }
202
203 void StartConnect() override {
204 std::lock_guard guard(Lock_);
205 Activate();
206 }
207
208 void Drain() override {
209 std::lock_guard guard(Lock_);
210 Activate();
211 if (Current_->Size == 0) {
212 return;
213 }
214 auto next = TakeSpare();
215 next->Size = 0;
216 SealedBytes_ += Current_->Size;
217 Current_.swap(next);
218 if (Queue_.Push(std::move(next))) {
219 Notifier_.Notify();
220 }
221 }

237 // Frames committed between two Drain calls. The buffer keeps its memory while the
238 // segment goes around: producer, Queue_, I/O side, Returned_, Spare_, producer.
239 struct TSegment : TMpscQueueNode<TSegment> {
240 // Counts itself: Segments() is what the node really holds, whatever path a segment took
241 explicit TSegment(std::atomic<size_t>& live)
242 : Live(live)
243 {
244 Live.fetch_add(1, std::memory_order_relaxed);
245 }
246
247 ~TSegment() {
248 Live.fetch_sub(1, std::memory_order_relaxed);
249 }
250
251 TSegment(const TSegment&) = delete;
252 TSegment& operator=(const TSegment&) = delete;
253
254 std::atomic<size_t>& Live;
255 // Not zero-filled: every byte below Size is written before the segment is sent, the
256 // padding by the producer and the payload holes by WritePayloads
257 TOutputBuffer Data;
258 size_t Size = 0;
259 // In the order of their frames; empty again once the I/O side has written them
260 std::vector<TDeferredPayload> Deferred;
261 };

263 // AlignState_ is the stream base shifted left by one, with this bit set once
264 // a frame with PayloadAlignment has been written
265 static constexpr uint64_t AlignedBit = 1;
266 static constexpr size_t MaxBytesBeforeYield = 2 * 1024 * 1024;

268 // Producer side
269 void Activate() {
270 if (!Active_) {
271 Active_ = true;
272 // Release pairs with the acquire load in Serve(): the notification alone
273 // tells the I/O side to look, not what to see.
274 ConnectRequested_.store(true, std::memory_order_release);
275 Notifier_.Notify();
276 }
277 }
278
279 // Producer side. The segment returned last comes first: a backlog that has passed
280 // does not send every spare buffer around again.
281 std::unique_ptr<TSegment> TakeSpare() {
282 while (auto segment = Returned_.Pop()) {
283 Spare_.push_back(std::move(segment));
284 }
285 if (Spare_.empty()) {
286 return std::make_unique<TSegment>(Segments_);
287 }
288 auto segment = std::move(Spare_.back());
289 Spare_.pop_back();
290 SpareCount_.fetch_sub(1, std::memory_order_relaxed);
291 return segment;
292 }
293
294 // I/O side. Only this side adds to SpareCount_, so the check keeps it at the cap.
295 // Relaxed: the count orders nothing, the segment itself goes through Returned_.
296 void Recycle(std::unique_ptr<TSegment> segment) {
297 if (SpareCount_.load(std::memory_order_relaxed) < MaxSpareSegments) {
298 SpareCount_.fetch_add(1, std::memory_order_relaxed);
299 Returned_.Push(std::move(segment));
300 }
301 }
302
303 // I/O side. The objects die here: a recycled segment must not bring them back.
304 void WritePayloads(TSegment& segment) noexcept {
305 for (const auto& payload : segment.Deferred) {
306 payload.Write(payload.Blob, std::span<char>(segment.Data.Data() + payload.Offset, payload.Size));
307 }
308 segment.Deferred.clear();
309 }

311 // I/O side. A failed wait is logged and tried again after the back-off: the node
312 // then looks for work once a second instead of never.
313 TFuture<void> WaitForWork() {
314 try {
315 co_await Notifier_.Wait();
316 co_return;
317 } catch (const std::exception& ex) {
318 std::cerr << "Error waiting for frames to " << HostPort_.ToString() << ": " << ex.what() << "\n";
319 }
320 co_await Poller_.Sleep(std::chrono::milliseconds(1000));
321 }
322
323 // I/O side
324 TFuture<void> Serve() {
325 // Like TNode, the node does not connect before StartConnect() or Drain()
326 while (!ConnectRequested_.load(std::memory_order_acquire)) {
327 co_await WaitForWork();
328 }
329 while (true) {
330 if (!Connected_) {
331 co_await Connect();
332 }
333 Writing_ = Queue_.Pop();
334 if (!Writing_) {
335 co_await WaitForWork();
336 continue;
337 }
338 // Counted before the write: a failed write drops the rest of the segment, as in TNode
339 SentBytes_ += Writing_->Size;
340 WritePayloads(*Writing_);
341 try {
342 co_await TByteWriter(Socket_).Write(Writing_->Data.Data(), Writing_->Size);
343 } catch (const std::exception& ex) {
344 std::cerr << "Error during draining: " << ex.what() << "\n";
345 Connected_ = false;
346 }
347 WrittenBytes_ += Writing_->Size;
348 Recycle(std::move(Writing_));
349 // A write that does not block does not suspend: without this a busy producer
350 // would keep the loop from its other coroutines. Not Yield(): resumed from
351 // a timer, it is resumed again in the same pass, before the poller polls.
352 if (WrittenBytes_ >= MaxBytesBeforeYield) {
353 WrittenBytes_ = 0;
354 co_await Poller_.Sleep(std::chrono::nanoseconds(1));
355 }
356 }
357 }

359 // I/O side
360 TFuture<void> Connect() {
361 // Without aligned frames the new stream starts at 0 with no filler, as TNode does:
362 // the base moves to the first byte of the new stream. Once the bit is set the base
363 // stays, and the filler below gives the queued frames their old offsets.
364 uint64_t state = AlignState_.load(std::memory_order_relaxed);
365 while (!(state & AlignedBit) && !AlignState_.compare_exchange_weak(state, SentBytes_ << 1,
366 std::memory_order_relaxed, std::memory_order_relaxed))
367 {
368 }
369 const uint64_t offset = (state & AlignedBit) ? SentBytes_ - (state >> 1) : 0;
370 std::cout << "Connecting to " << HostPort_.ToString() << "\n";
371 while (!Connected_) {
372 try {
373 TAddress addr = co_await HostPort_.Resolve(Resolver_);
374 Socket_ = SocketFactory_(addr);
375 // Small messages (acks) must not wait for Nagle and delayed ACK.
376 Socket_.SetNoDelay();
377 co_await Socket_.Connect(addr);
378 char filler[MaxFillerSize];
379 if (const size_t size = WriteFiller(filler, offset)) {
380 co_await TByteWriter(Socket_).Write(filler, size);
381 }
382 Connected_ = true;
383 std::cout << "Connected to " << HostPort_.ToString() << "\n";
384 } catch (const std::exception& ex) {
385 std::cerr << "Error connecting to " << HostPort_.ToString() << ": " << ex.what() << "\n";
386 Connected_ = false;
387 }
388 if (!Connected_) {
389 co_await Poller_.Sleep(std::chrono::milliseconds(1000));
390 }
391 }
392 }
