128 // The framing is the one of TNode::Send; a deferred payload is left to the I/O side
129 void Send(TEnvelope&& envelope) override {
130 // Empty blobs skip the factory and its compile-time check of the id
131 if (envelope.MessageId > MaxMessageId) [[unlikely]] {
132 throw std::invalid_argument("message id " + std::to_string(envelope.MessageId)
133 + " overlaps the padding bits of THeader::MessageId");
134 }
135 std::lock_guard guard(Lock_);
136 auto blob = std::move(envelope.Blob);
137 uint32_t padding = 0;
138 uint32_t payload = 0;
139 const TMessagesFactory::TPayloadWriter* writer = nullptr;
140 if (blob.Size > 0) {
141 if (blob.Type == TBlob::PointerType::Near) {
142 writer = Factory_.PayloadWriter(envelope.MessageId);
143 }
144 if (writer) {
145 // Sized here: an invalid object throws to the sender, as with TNode
146 payload = writer->Size(blob);
147 } else {
148 blob = Factory_.SerializeFar(envelope.MessageId, std::move(blob));
149 payload = blob.Size;
150 }
151 const size_t alignment = Factory_.PayloadAlignment(envelope.MessageId);
152 if (alignment > 1) {
153 padding = PayloadPadding(Offset(), alignment);
154 }
155 }
156 THeader data{
157 .Sender = envelope.Sender,
158 .Recipient = envelope.Recipient,
159 .MessageId = envelope.MessageId | (padding << HeaderPaddingShift),
160 .Size = payload
161 };
162 const size_t size = sizeof(data) + padding + payload;
163 auto buf = Acquire(size);
164 std::memcpy(buf.data(), &data, sizeof(data));
165 std::memset(buf.data() + sizeof(data), 0, padding);
166 if (writer) {
167 Current_->Deferred.push_back(TDeferredPayload{
168 .Offset = Current_->Size + sizeof(data) + padding,
169 .Size = payload,
170 .Write = writer->Write,
171 .Blob = std::move(blob),
172 });
173 } else if (payload > 0) {
174 std::memcpy(buf.data() + sizeof(data) + padding, blob.Data.get(), payload);
175 }
176 Commit(size);
177 }
178
179 std::span<char> Acquire(size_t size) override {
180 auto& segment = *Current_;
181 if (segment.Size + size > segment.Data.size()) {
182 segment.Data.resize(segment.Size + size);
183 }
184 return std::span<char>(segment.Data.data() + segment.Size, size);
185 }
186
187 void Commit(size_t size) override {
188 Current_->Size += size;
189 }
190
191 uint64_t Offset() const override {
192 if (!WroteAligned_) [[unlikely]] {
193 // One atomic step against the CAS in Connect(): either that moved the base first
194 // and this reads the new one, or it sees the bit and keeps the base. The word
195 // carries all the two sides share here, so nothing is ordered by it: relaxed.
196 Base_ = AlignState_.fetch_or(AlignedBit, std::memory_order_relaxed) >> 1;
197 WroteAligned_ = true;
198 }
199 return SealedBytes_ + Current_->Size - Base_;
200 }
201
202 void StartConnect() override {
203 std::lock_guard guard(Lock_);
204 Activate();
205 }
206
207 void Drain() override {
208 std::lock_guard guard(Lock_);
209 Activate();
210 if (Current_->Size == 0) {
211 return;
212 }
213 auto next = TakeSpare();
214 next->Size = 0;
215 SealedBytes_ += Current_->Size;
216 Current_.swap(next);
217 if (Queue_.Push(std::move(next))) {
218 Notifier_.Notify();
219 }
220 }

222 THostPort GetHostPort() const override {
223 return HostPort_;

236 // Frames committed between two Drain calls. The buffer keeps its memory while the
237 // segment goes around: producer, Queue_, I/O side, Returned_, Spare_, producer.
238 struct TSegment : TMpscQueueNode<TSegment> {
239 // Counts itself: Segments() is what the node really holds, whatever path a segment took
240 explicit TSegment(std::atomic<size_t>& live)
241 : Live(live)
242 {
243 Live.fetch_add(1, std::memory_order_relaxed);
244 }
245
246 ~TSegment() {
247 Live.fetch_sub(1, std::memory_order_relaxed);
248 }
249
250 TSegment(const TSegment&) = delete;
251 TSegment& operator=(const TSegment&) = delete;
252
253 std::atomic<size_t>& Live;
254 std::vector<char> Data;
255 size_t Size = 0;
256 // In the order of their frames; empty again once the I/O side has written them
257 std::vector<TDeferredPayload> Deferred;
258 };

260 // AlignState_ is the stream base shifted left by one, with this bit set once
261 // a frame with PayloadAlignment has been written
262 static constexpr uint64_t AlignedBit = 1;
263 static constexpr size_t MaxBytesBeforeYield = 2 * 1024 * 1024;
264
265 // Producer side
266 void Activate() {
267 if (!Active_) {
268 Active_ = true;
269 // Release pairs with the acquire load in Serve(): the notification alone
270 // tells the I/O side to look, not what to see.
271 ConnectRequested_.store(true, std::memory_order_release);
272 Notifier_.Notify();
273 }
274 }

276 // Producer side. The segment returned last comes first: a backlog that has passed
277 // does not send every spare buffer around again.
278 std::unique_ptr<TSegment> TakeSpare() {
279 while (auto segment = Returned_.Pop()) {
280 Spare_.push_back(std::move(segment));
281 }
282 if (Spare_.empty()) {
283 return std::make_unique<TSegment>(Segments_);
284 }
285 auto segment = std::move(Spare_.back());
286 Spare_.pop_back();
287 SpareCount_.fetch_sub(1, std::memory_order_relaxed);
288 return segment;
289 }
290
291 // I/O side. Only this side adds to SpareCount_, so the check keeps it at the cap.
292 // Relaxed: the count orders nothing, the segment itself goes through Returned_.
293 void Recycle(std::unique_ptr<TSegment> segment) {
294 if (SpareCount_.load(std::memory_order_relaxed) < MaxSpareSegments) {
295 SpareCount_.fetch_add(1, std::memory_order_relaxed);
296 Returned_.Push(std::move(segment));
297 }
298 }
299
300 // I/O side. The objects die here: a recycled segment must not bring them back.
301 void WritePayloads(TSegment& segment) noexcept {
302 for (const auto& payload : segment.Deferred) {
303 payload.Write(payload.Blob, std::span<char>(segment.Data.data() + payload.Offset, payload.Size));
304 }
305 segment.Deferred.clear();
306 }

308 // I/O side. A failed wait is logged and tried again after the back-off: the node
309 // then looks for work once a second instead of never.
310 TFuture<void> WaitForWork() {
311 try {
312 co_await Notifier_.Wait();
313 co_return;
314 } catch (const std::exception& ex) {
315 std::cerr << "Error waiting for frames to " << HostPort_.ToString() << ": " << ex.what() << "\n";
316 }
317 co_await Poller_.Sleep(std::chrono::milliseconds(1000));
318 }
319
320 // I/O side
321 TFuture<void> Serve() {
322 // Like TNode, the node does not connect before StartConnect() or Drain()
323 while (!ConnectRequested_.load(std::memory_order_acquire)) {
324 co_await WaitForWork();
325 }
326 while (true) {
327 if (!Connected_) {
328 co_await Connect();
329 }
330 Writing_ = Queue_.Pop();
331 if (!Writing_) {
332 co_await WaitForWork();
333 continue;
334 }
335 // Counted before the write: a failed write drops the rest of the segment, as in TNode
336 SentBytes_ += Writing_->Size;
337 WritePayloads(*Writing_);
338 try {
339 co_await TByteWriter(Socket_).Write(Writing_->Data.data(), Writing_->Size);
340 } catch (const std::exception& ex) {
341 std::cerr << "Error during draining: " << ex.what() << "\n";
342 Connected_ = false;
343 }
344 WrittenBytes_ += Writing_->Size;
345 Recycle(std::move(Writing_));
346 // A write that does not block does not suspend: without this a busy producer
347 // would keep the loop from its other coroutines. Not Yield(): resumed from
348 // a timer, it is resumed again in the same pass, before the poller polls.
349 if (WrittenBytes_ >= MaxBytesBeforeYield) {
350 WrittenBytes_ = 0;
351 co_await Poller_.Sleep(std::chrono::nanoseconds(1));
352 }
353 }
354 }
355
356 // I/O side
357 TFuture<void> Connect() {
358 // Without aligned frames the new stream starts at 0 with no filler, as TNode does:
359 // the base moves to the first byte of the new stream. Once the bit is set the base
360 // stays, and the filler below gives the queued frames their old offsets.
361 uint64_t state = AlignState_.load(std::memory_order_relaxed);
362 while (!(state & AlignedBit) && !AlignState_.compare_exchange_weak(state, SentBytes_ << 1,
363 std::memory_order_relaxed, std::memory_order_relaxed))
364 {
365 }
366 const uint64_t offset = (state & AlignedBit) ? SentBytes_ - (state >> 1) : 0;
367 std::cout << "Connecting to " << HostPort_.ToString() << "\n";
368 while (!Connected_) {
369 try {
370 TAddress addr = co_await HostPort_.Resolve(Resolver_);
371 Socket_ = SocketFactory_(addr);
372 // Small messages (acks) must not wait for Nagle and delayed ACK.
373 Socket_.SetNoDelay();
374 co_await Socket_.Connect(addr);
375 char filler[MaxFillerSize];
376 if (const size_t size = WriteFiller(filler, offset)) {
377 co_await TByteWriter(Socket_).Write(filler, size);
378 }
379 Connected_ = true;
380 std::cout << "Connected to " << HostPort_.ToString() << "\n";
381 } catch (const std::exception& ex) {
382 std::cerr << "Error connecting to " << HostPort_.ToString() << ": " << ex.what() << "\n";
383 Connected_ = false;
384 }
385 if (!Connected_) {
386 co_await Poller_.Sleep(std::chrono::milliseconds(1000));
387 }
388 }
389 }
