76 bool Push(std::unique_ptr<T> item) noexcept {
77 T* node = item.release();
78 // The value is only stored into node->Next, and the CAS checks that it is still the head.
79 T* head = Head_.load(std::memory_order_relaxed);
80 do {
81 node->Next = head;
82 // Release publishes the item and its link to the acquire exchange in Pop.
83 // The head is changed only by CAS and exchange, so that exchange is also
84 // ordered after the pushes of the items below (release sequence).
85 } while (!Head_.compare_exchange_weak(head, node, std::memory_order_release, std::memory_order_relaxed));
86 return head == nullptr;
87 }

92 std::unique_ptr<T> Pop() noexcept {
93 if (!Ready_) {
94 // An idle consumer must not take the cache line of the head from the producers.
95 // A push that this load misses found the stack empty and notifies.
96 if (!Head_.load(std::memory_order_relaxed)) {
97 return nullptr;
98 }
99 // Acquire pairs with the release CAS in Push.
100 T* node = Head_.exchange(nullptr, std::memory_order_acquire);
101 while (node) {
102 T* next = node->Next;
103 node->Next = Ready_;
104 Ready_ = node;
105 node = next;
106 }
107 }
108 T* node = Ready_;
109 if (!node) {
110 return nullptr;
111 }
112 Ready_ = node->Next;
113 node->Next = nullptr;
114 return std::unique_ptr<T>(node);
115 }
116
117private:
118 // Newest first; shared with the producers
119 std::atomic<T*> Head_{nullptr};
120 // Oldest first; taken from Head_, used only by the consumer
121 T* Ready_ = nullptr;
122};
