73 bool Full() const {
74 return Queued_.load(std::memory_order_relaxed) >= Limit_;
75 }

78 void Deliver(std::unique_ptr<TEnvelopeBatch> batch) {
79 // Relaxed: the count orders nothing, the batch itself goes through Queue_
80 Queued_.fetch_add(batch->Bytes, std::memory_order_relaxed);
81 if (Queue_.Push(std::move(batch))) {
82 Notifier_->Notify();
83 }
84 }
85
87 size_t QueuedBytes() const {
88 return Queued_.load(std::memory_order_relaxed);
89 }
90
91private:
92 static constexpr size_t MaxBytesBeforeYield = 2 * 1024 * 1024;
93
94 // Actor side. A failed wait is logged and tried again after a second, as in TThreadedNode.
95 TFuture<void> WaitForWork() {
96 try {
97 co_await Notifier_->Wait();
98 co_return;
99 } catch (const std::exception& ex) {
100 std::cerr << "Error waiting for inbound envelopes: " << ex.what() << "\n";
101 }
102 co_await System_.Sleep(std::chrono::milliseconds(1000));
103 }
104
105 // Actor side
106 TFuture<void> Consume() {
107 size_t delivered = 0;
108 while (true) {
109 auto batch = Queue_.Pop();
110 if (!batch) {
111 co_await WaitForWork();
112 continue;
113 }
114 for (auto& envelope : batch->Envelopes) {
115 System_.Send(envelope.Sender, envelope.Recipient, envelope.MessageId, std::move(envelope.Blob));
116 }
117 Queued_.fetch_sub(batch->Bytes, std::memory_order_relaxed);
118 delivered += batch->Bytes;
119 // Not Yield(): resumed from a timer, it is resumed again in the same pass (issue 244)
120 if (delivered >= MaxBytesBeforeYield) {
121 delivered = 0;
122 co_await System_.Sleep(std::chrono::nanoseconds(1));
123 }
124 }
125 }
126
127 TActorSystem& System_;
128 const size_t Limit_;
129 std::unique_ptr<INotifier> Notifier_;
130 TMpscQueue<TEnvelopeBatch> Queue_;
131 std::atomic<size_t> Queued_ = 0;
132 TFuture<void> Consumer_;
133};

252 TFuture<void> Accept() {
253 while (true) {
254 auto client = co_await Listener_.Accept();
255 if (client) {
256 std::erase_if(Connections_, [](const auto& connection) {
257 return connection->Reader.done();
258 });
259 client->SetNoDelay();
260 auto& connection = *Connections_.emplace_back(std::make_unique<TConnection>(std::move(*client)));
261 connection.Reader = Read(connection.Socket);
262 } else if (!IsTransientAcceptError(client.error())) {
263 if (Logger_) {
264 Logger_("Accept failed: " + std::system_category().message(client.error()));
265 }
266 co_await Poller_.Sleep(std::chrono::milliseconds(100));
267 }
268 }

271 // False if the deserializer throws: the envelope is then dropped
272 bool Deserialize(TEnvelope& envelope) {
273 try {
274 envelope.Blob = Factory_->DeserializeFar(envelope.MessageId, std::move(envelope.Blob));
275 return true;
276 } catch (const std::exception& e) {
277 if (Logger_) {

279 + " for " + envelope.Recipient.ToString() + ": " + e.what());

285 TFuture<void> Read(TSocket& socket) {
286 TEnvelopeReader envelopeReader(InflightBytes, /*lowWatermark = */ 1024);
287 size_t bytesRead = 0;
288 try {
289 while (true) {
290 while (Delivery_.Full()) {
291 co_await Poller_.Sleep(std::chrono::milliseconds(1));
292 }
293 auto buffer = envelopeReader.Acquire(ReadSize);
294 auto size = co_await socket.ReadSome(buffer.data(), buffer.size());
295 if (size < 0) {
296 continue;
297 }
298 if (size == 0) {
299 throw std::runtime_error("Socket closed");
300 }
301 envelopeReader.Commit(size);
302
303 std::unique_ptr<TEnvelopeBatch> batch;
304 while (auto envelope = envelopeReader.Pop()) {
305 if (envelope->Recipient.NodeId() != NodeId_) [[unlikely]] {
306 if (Logger_) {
307 Logger_("Received message for different node: " + envelope->Recipient.ToString());
308 }
309 continue;
310 }
311 const size_t bytes = envelope->Blob.Size + sizeof(THeader);
312 if (Factory_ && !Deserialize(*envelope)) {
313 continue;
314 }
315 if (!batch) {
316 batch = std::make_unique<TEnvelopeBatch>();
317 }
318 batch->Bytes += bytes;
319 batch->Envelopes.push_back(std::move(*envelope));
320 }
321 if (batch) {
322 Delivery_.Deliver(std::move(batch));
323 }
324
325 // A read that does not block does not suspend: without this a busy peer
326 // would keep the loop from the other connections. Not Yield(), see TThreadedNode.
327 bytesRead += size;
328 if (bytesRead >= MaxBytesBeforeYield) {
329 bytesRead = 0;
330 co_await Poller_.Sleep(std::chrono::nanoseconds(1));
331 }
332 }
333 } catch (const std::exception& e) {
334 if (Logger_) {
335 Logger_(std::string("Error in InboundConnection: ") + e.what());
336 }
337 }
338 socket.Close();
339 }
