24struct THeaderNameLess {
25 using is_transparent = void;
26 bool operator()(std::string_view a, std::string_view b) const {
27 return std::lexicographical_compare(a.begin(), a.end(), b.begin(), b.end(), [](char x, char y) {
28 return Lower(x) < Lower(y);
29 });
30 }
31private:
32 static char Lower(char c) {
33 return (c >= 'A' && c <= 'Z') ? char(c - 'A' + 'a') : c;
34 }

307 TVoidTask HandleClient(TSocket clientSocket) {
308 auto byteReader = TByteReader<TSocket>(clientSocket);
309
310 auto bodyReader = [&](char* buffer, size_t size) -> TFuture<ssize_t> {
311 co_return co_await byteReader.ReadSome(buffer, size);
312 };
313
314 auto chunkHeaderReader = [&]() -> TFuture<std::string> {
315 co_return co_await byteReader.ReadUntil("\r\n");
316 };
317
318 auto bodyWriter = [&](const void* data, size_t size) -> TFuture<ssize_t> {
319 co_return co_await clientSocket.WriteSome(data, size);
320 };
321
322 auto hangUpWaiter = [&]() -> TFuture<void> {
323 co_await clientSocket.Monitor();
324 };
325
326 std::string clientString = clientSocket.RemoteAddr() ? clientSocket.RemoteAddr()->ToString() : "unknown";
327
328 try {
329 while (true) {
330 auto header = co_await byteReader.ReadUntil("\r\n\r\n");
331 TRequest request(std::move(header), bodyReader, chunkHeaderReader, hangUpWaiter);
332 TResponse response(bodyWriter);
333 co_await Router.HandleRequest(request, response);
334 Log(request, response, clientString);
335 if (response.IsClosed() || request.RequireConnectionClose()) {
336 break;
337 }
338 }
339 } catch (const std::exception& ex) {
340 if (Logger) {
341 Logger(std::string("Client handler exception: ") + ex.what());
342 }
343 }
344 co_return;
345 }

347 void Log(const TRequest& request, const TResponse& response, const std::string& clientString) {
348 if (!Logger) {
349 return;
350 }
351 LogStream.str("");
352 LogStream.clear();
353
354 // Build full path with query parameters if any
355 std::string fullPath = request.Uri().Path();
356 const auto& qp = request.Uri().QueryParameters();
357 if (!qp.empty()) {
358 fullPath.push_back('?');
359 bool first = true;
360 for (const auto& [k,v] : qp) {
361 if (!first) fullPath.push_back('&');
362 first = false;
363 fullPath.append(k);
364 fullPath.push_back('=');
365 fullPath.append(v);
366 }
367 }
368
369 // Timestamp in nginx style: [08/Nov/2025:15:23:10 +0100]
370 auto now = std::chrono::system_clock::now();
371 std::time_t raw = std::chrono::system_clock::to_time_t(now);
372 std::tm localTm{};
373 std::tm gmTm{};
374#if defined(_WIN32)
375 localtime_s(&localTm, &raw);
376 gmtime_s(&gmTm, &raw);
377#else
378 localtime_r(&raw, &localTm);
379 gmtime_r(&raw, &gmTm);
380#endif
381 // Compute timezone offset
382 // mktime converts tm in local time to time_t; difference gives offset vs UTC
383 std::time_t localTime = std::mktime(&localTm);
384 std::time_t gmTime = std::mktime(&gmTm);
385 long tzOffsetSec = static_cast<long>(difftime(localTime, gmTime));
386 int tzSign = tzOffsetSec >= 0 ? 1 : -1;
387 tzOffsetSec = std::labs(tzOffsetSec);
388 int tzHours = static_cast<int>(tzOffsetSec / 3600);
389 int tzMins = static_cast<int>((tzOffsetSec % 3600) / 60);
390 char tzBuf[8];
391 std::snprintf(tzBuf, sizeof(tzBuf), "%c%02d%02d", tzSign > 0 ? '+' : '-', tzHours, tzMins);
392
393 // Prepare HTTP version string
394 std::string_view ver = request.Version();
395 bool hasHttpPrefix = ver.size() >= 5 && ver.substr(0,5) == "HTTP/";
396
397 LogStream << clientString << " - - [" << std::put_time(&localTm, "%d/%b/%Y:%H:%M:%S ") << tzBuf << "] \""
398 << request.Method() << ' ' << fullPath << ' ' << (hasHttpPrefix ? std::string(ver) : (std::string("HTTP/") + std::string(ver))) << "\" "
399 << response.StatusCode() << ' ' << '-' << ' ' << "\"-\" \"-\"";
400 Logger(LogStream.str());
401 }
