66 bool Carry() {
67 size_t size = 0;
68 for (auto* op : Ready_) {
69 if (op) {
70 op->Index = static_cast<unsigned>(size);
71 Ready_[size++] = op;
72 }
73 }
74 Ready_.resize(size);
75 for (auto* op : Reaped_) {
76 if (op) {
77 Add(Ready_, op);
78 }
79 }
80 Reaped_.clear();
81 return !Ready_.empty();
82 }
83
85 void Wakeup() {
86 for (size_t i = 0; i < Ready_.size(); i++) {
87 // Taken out before resuming: the coroutine may post its next operation
88 // at the same address, and Drop() must not find this entry for it.
89 if (auto* op = std::exchange(Ready_[i], nullptr)) {
90 std::exchange(op->Handle, {}).resume();
91 }
92 }
93 Ready_.clear();
94 }
95
96private:
97 static void Add(std::vector<TOperationState*>& list, TOperationState* op) {
98 op->Index = static_cast<unsigned>(list.size());
99 list.push_back(op);
100 }
101
102 static bool Remove(std::vector<TOperationState*>& list, TOperationState& op) {
103 if (op.Index < list.size() && list[op.Index] == &op) {
104 list[op.Index] = nullptr;
105 return true;
106 }
107 return false;
108 }
