289 void Wakeup(TEvent&& change) {
307 // Removed or closed since the event was collected.
308 if (!change.Handle || Removing(change.Fd)) {
309 return;
310 }
311 auto index = Changes_.size();
312 change.Handle.resume();
313 // A removal queued during the resume already removes everything.
314 if (change.Fd >= 0 && !Removing(change.Fd)) {
315 bool matched = false;
316 for (; index < Changes_.size() && !matched; index++) {
317 matched = Changes_[index].Match(change);
318 }
319 if (!matched) {
320 change.Handle = {};
321 Changes_.emplace_back(std::move(change));
322 }
323 }
324 }

385 TEvent Resolve(const TEvent& change, const THandlePair& ev) const {
386 assert(&change >= Changes_.data() && &change < Changes_.data() + Changes_.size());
387 if (!(change.Type & TEvent::CLOSE) && Removing(change.Fd)
388 && &change - Changes_.data() < Removal_[change.Fd])
389 {
390 return TEvent{change.Fd, 0, {}};
391 }
392 if (change.Type != TEvent::CANCEL || !change.Handle) {
393 return change;
394 }
395 int type = 0;
396 if (ev.Read == change.Handle) {
397 type |= TEvent::READ;
398 }
399 if (ev.Write == change.Handle) {
400 type |= TEvent::WRITE;
401 }
402 if (ev.RHup == change.Handle) {
403 type |= TEvent::RHUP;
404 }
405 return TEvent{change.Fd, type, {}};
406 }

429 void ProcessTimers() {
430 auto now = TClock::now();
431 bool first = true;
432 unsigned prevId = 0;
433
434 while (!Timers_.empty()&&Timers_.top().Deadline <= now) {
435 TTimer timer = Timers_.top(); Timers_.pop();
436
437 if ((first || prevId != timer.Id) && timer.Handle) { // skip removed timers
438 LastFiredTimer_ = timer.Id;
439 timer.Handle.resume();
440 }
441
442 first = false;
443 prevId = timer.Id;
444 }
445
446 LastTimersProcessTime_ = now;
447 }

461 void QueueRemoval(int fd, int close) {
462 assert(fd >= 0);
463 MaxFd_ = std::max(MaxFd_, fd);
464 if (fd >= static_cast<int>(Removal_.size())) {
465 Removal_.resize(fd + 1, -1);
466 }
467 Removal_[fd] = static_cast<int>(Changes_.size());
468 Changes_.emplace_back(TEvent{fd, TEvent::READ|TEvent::WRITE|TEvent::RHUP|close, {}});
469 }
470 bool Removing(int fd) const {
471 return fd >= 0 && fd < static_cast<int>(Removal_.size()) && Removal_[fd] >= 0;
472 }
