302 void Wakeup(TEvent&& change) {
320 // Removed or closed since the event was collected.
321 if (!change.Handle || Removing(change.Fd)) {
322 return;
323 }
324 auto index = Changes_.size();
325 change.Handle.resume();
326 // A removal queued during the resume already removes everything.
327 if (change.Fd >= 0 && !Removing(change.Fd)) {
328 bool matched = false;
329 for (; index < Changes_.size() && !matched; index++) {
330 matched = Changes_[index].Match(change);
331 }
332 if (!matched) {
333 change.Handle = {};
334 Changes_.emplace_back(std::move(change));
335 }
336 }
337 }

398 TEvent Resolve(const TEvent& change, const THandlePair& ev) const {
399 assert(&change >= Changes_.data() && &change < Changes_.data() + Changes_.size());
400 if (!(change.Type & TEvent::CLOSE) && Removing(change.Fd)
401 && &change - Changes_.data() < Removal_[change.Fd])
402 {
403 return TEvent{change.Fd, 0, {}};
404 }
405 if (change.Type != TEvent::CANCEL || !change.Handle) {
406 return change;
407 }
408 int type = 0;
409 if (ev.Read == change.Handle) {
410 type |= TEvent::READ;
411 }
412 if (ev.Write == change.Handle) {
413 type |= TEvent::WRITE;
414 }
415 if (ev.RHup == change.Handle) {
416 type |= TEvent::RHUP;
417 }
418 return TEvent{change.Fd, type, {}};
419 }

442 void ProcessTimers() {
443 auto now = TClock::now();
444 bool first = true;
445 unsigned prevId = 0;
446
447 while (!Timers_.empty()&&Timers_.top().Deadline <= now) {
448 TTimer timer = Timers_.top(); Timers_.pop();
449
450 if ((first || prevId != timer.Id) && timer.Handle) { // skip removed timers
451 LastFiredTimer_ = timer.Id;
452 timer.Handle.resume();
453 }
454
455 first = false;
456 prevId = timer.Id;
457 }
458
459 LastTimersProcessTime_ = now;
460 }

474 void QueueRemoval(int fd, int close) {
475 assert(fd >= 0);
476 MaxFd_ = std::max(MaxFd_, fd);
477 if (fd >= static_cast<int>(Removal_.size())) {
478 Removal_.resize(fd + 1, -1);
479 }
480 Removal_[fd] = static_cast<int>(Changes_.size());
481 Changes_.emplace_back(TEvent{fd, TEvent::READ|TEvent::WRITE|TEvent::RHUP|close, {}});
482 }
483 bool Removing(int fd) const {
484 return fd >= 0 && fd < static_cast<int>(Removal_.size()) && Removal_[fd] >= 0;
485 }
