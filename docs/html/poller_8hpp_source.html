271 void Wakeup(TEvent&& change) {
289 if (!change.Handle) {
290 return;
291 }
292 auto index = Changes_.size();
293 change.Handle.resume();
294 if (change.Fd >= 0) {
295 bool matched = false;
296 for (; index < Changes_.size() && !matched; index++) {
297 matched = Changes_[index].Match(change);
298 }
299 if (!matched) {
300 change.Handle = {};
301 Changes_.emplace_back(std::move(change));
302 }
303 }
304 }

394 void ProcessTimers() {
395 auto now = TClock::now();
396 bool first = true;
397 unsigned prevId = 0;
398
399 while (!Timers_.empty()&&Timers_.top().Deadline <= now) {
400 TTimer timer = Timers_.top(); Timers_.pop();
401
402 if ((first || prevId != timer.Id) && timer.Handle) { // skip removed timers
403 LastFiredTimer_ = timer.Id;
404 timer.Handle.resume();
405 }
406
407 first = false;
408 prevId = timer.Id;
409 }
410
411 LastTimersProcessTime_ = now;
412 }
